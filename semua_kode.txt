package com.example.sembakoapi.web;

import com.example.sembakoapi.model.Product;
import com.example.sembakoapi.repository.CategoryRepository;
import com.example.sembakoapi.repository.ProductRepository;
import com.example.sembakoapi.repository.SupplierRepository;
import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;

@RestController
@RequestMapping("/api/products")
public class ProductController {
  private final ProductRepository products;
  private final CategoryRepository categories;
  private final SupplierRepository suppliers;

  public ProductController(ProductRepository products, CategoryRepository categories, SupplierRepository suppliers) {
    this.products = products; this.categories = categories; this.suppliers = suppliers;
  }

  @GetMapping
  public List<Product> list() {
    List<Product> all = products.findAll();
    all.sort((a, b) -> a.name.compareToIgnoreCase(b.name));
    all.forEach(this::withJoinedNames);
    return all;
  }

  @PostMapping
  public Product create(@RequestBody Product body) {
    body.id = null;
    if (body.name == null || body.name.isBlank()) throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Nama wajib diisi");
    return withJoinedNames(products.save(body));
  }

  @PutMapping("/{id}")
  public Product update(@PathVariable Long id, @RequestBody Product body) {
    if (body.name == null || body.name.isBlank()) throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Nama wajib diisi");
    Product existing = products.findById(id).orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND));
    existing.name = body.name; existing.categoryId = body.categoryId; existing.supplierId = body.supplierId;
    existing.price = body.price; existing.stock = body.stock; existing.minStock = body.minStock; existing.unit = body.unit;
    return withJoinedNames(products.save(existing));
  }

  @DeleteMapping("/{id}")
  @ResponseStatus(HttpStatus.NO_CONTENT)
  public void delete(@PathVariable Long id) {
    if (!products.existsById(id)) throw new ResponseStatusException(HttpStatus.NOT_FOUND);
    products.deleteById(id);
  }

  private Product withJoinedNames(Product p) {
    p.categoryName = p.categoryId == null ? null : categories.findById(p.categoryId).map(c -> c.name).orElse(null);
    p.supplierName = p.supplierId == null ? null : suppliers.findById(p.supplierId).map(s -> s.name).orElse(null);
    return p;
  }
}

    s.totalCategories = (int) categories.count();

        });

import android.app.Application;
import android.os.Handler;
import android.os.Looper;
import androidx.annotation.NonNull;
import androidx.lifecycle.AndroidViewModel;
import androidx.lifecycle.LiveData;
import androidx.lifecycle.MutableLiveData;
import com.example.sembakoapp.R;
import com.example.sembakoapp.data.SembakoException;
import com.example.sembakoapp.data.SembakoRepository;
import com.example.sembakoapp.data.ServiceLocator;
import com.example.sembakoapp.data.model.Category;
import com.example.sembakoapp.data.model.Product;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class CategoriesViewModel extends AndroidViewModel {

  private final SembakoRepository repository = ServiceLocator.repository();
  private final ExecutorService executor = Executors.newSingleThreadExecutor();
  private final Handler mainHandler = new Handler(Looper.getMainLooper());

  private final MutableLiveData<List<Category>> categories = new MutableLiveData<>();
  private final MutableLiveData<Map<Long, Integer>> productCounts = new MutableLiveData<>();
  private final MutableLiveData<String> message = new MutableLiveData<>();

  public CategoriesViewModel(@NonNull Application application) {
    super(application);
    refresh();
  }

  public LiveData<List<Category>> getCategories() {
    return categories;
  }

  public LiveData<Map<Long, Integer>> getProductCounts() {
    return productCounts;
  }

  public LiveData<String> getMessage() {
    return message;
  }

  public void clearMessage() {
    message.setValue(null);
  }

  public void refresh() {
    executor.execute(() -> {
      try {
        final List<Category> loadedCategories = repository.getCategories();
        final Map<Long, Integer> counts = new HashMap<>();
        for (Product p : repository.getProducts()) {
          Integer current = counts.get(p.categoryId);
          counts.put(p.categoryId, current == null ? 1 : current + 1);
        }
        mainHandler.post(() -> {
          categories.setValue(loadedCategories);
          productCounts.setValue(counts);
        });
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  public void save(Category category) {
    executor.execute(() -> {
      try {
        if (category.id == 0) repository.addCategory(category);
        else repository.updateCategory(category);
        mainHandler.post(this::refresh);
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  public void delete(Category category) {
    executor.execute(() -> {
      try {
        int used = 0;
        for (Product p : repository.getProducts()) {
          if (p.categoryId == category.id) used++;
        }
        if (used > 0) {
          final String text = getApplication().getString(R.string.category_in_use, used);
          mainHandler.post(() -> message.setValue(text));
          return;
        }
        repository.deleteCategory(category.id);
        mainHandler.post(this::refresh);
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  @Override
  protected void onCleared() {
    super.onCleared();
    executor.shutdownNow();
  }
}

  private final MutableLiveData<List<Product>> lowStock =

    return lowStock;
  }

  public LiveData<String> getMessage() {
    return message;
  }

  public void clearMessage() {
    message.setValue(null);
  }

  public void refresh() {
    executor.execute(() -> {
      try {
        final DashboardSummary loadedSummary = repository.getSummary();
        final List<Product> low = new ArrayList<>();
        for (Product p : repository.getProducts()) if (p.isLowStock()) low.add(p);
        Collections.sort(low, (a, b) -> Integer.compare(a.stock, b.stock));

        mainHandler.post(() -> {
          summary.setValue(loadedSummary);
          lowStock.setValue(low);
        });
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  @Override
  protected void onCleared() {
    super.onCleared();
    executor.shutdownNow();
  }

  public boolean isFilterActive() {
    return categoryFilter != 0 || supplierFilter != 0 || minStockFilter >= 0 || maxStockFilter >= 0;
  }

  public void refresh() {
    executor.execute(() -> {
      try {
        final List<Product> loadedProducts = repository.getProducts();
        final List<Category> loadedCategories = repository.getCategories();
        final List<Supplier> loadedSuppliers = repository.getSuppliers();

        mainHandler.post(() -> {
          allProducts = loadedProducts;
          categories.setValue(loadedCategories);
          suppliers.setValue(loadedSuppliers);
          applyFilters();
        });
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  public void setQuery(String value) {
    query = value == null ? "" : value.trim();
    applyFilters();
  }

  public void setFilters(long categoryId, long supplierId, int minStock, int maxStock) {
    categoryFilter = categoryId;
    supplierFilter = supplierId;
    minStockFilter = minStock;
    maxStockFilter = maxStock;
    applyFilters();
  }

  public void resetFilters() {
    setFilters(0, 0, -1, -1);
  }

  public void save(Product product) {
    executor.execute(() -> {
      try {
        if (product.id == 0) repository.addProduct(product);
        else repository.updateProduct(product);
        mainHandler.post(this::refresh);
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  public void delete(Product product) {
    executor.execute(() -> {
      try {
        repository.deleteProduct(product.id);
        mainHandler.post(this::refresh);
      } catch (SembakoException e) {
        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });
  }

  private void applyFilters() {
    List<Product> result = new ArrayList<>();
    for (Product p : allProducts) {
      if (!matchesQuery(p)) continue;
      if (categoryFilter != 0 && p.categoryId != categoryFilter) continue;
      if (supplierFilter != 0 && p.supplierId != supplierFilter) continue;
      if (minStockFilter >= 0 && p.stock < minStockFilter) continue;
      if (maxStockFilter >= 0 && p.stock > maxStockFilter) continue;
      result.add(p);
    }
    products.setValue(result);
  }

  private boolean matchesQuery(Product p) {
    if (query.isEmpty()) return true;
    String q = query.toLowerCase();
    return contains(p.name, q) || contains(p.categoryName, q) || contains(p.supplierName, q);
  }

  private boolean contains(String value, String query) {
    return value != null && value.toLowerCase().contains(query);
  }

  @Override
  protected void onCleared() {
    super.onCleared();
    executor.shutdownNow();
  }
}

        mainHandler.post(() -> message.setValue(e.getMessage()));
      }
    });

import java.util.List;
import okhttp3.mockwebserver.MockResponse;
import okhttp3.mockwebserver.MockWebServer;
import okhttp3.mockwebserver.RecordedRequest;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;
import retrofit2.Retrofit;
import retrofit2.converter.gson.GsonConverterFactory;

public class RemoteSembakoRepositoryTest {

  private MockWebServer server;
  private RemoteSembakoRepository repository;

  @Before
  public void setUp() throws Exception {
    server = new MockWebServer();
    server.start();
    SembakoApiService api =
        new Retrofit.Builder()
            .baseUrl(server.url("/"))
            .addConverterFactory(GsonConverterFactory.create())
            .build()
            .create(SembakoApiService.class);
    repository = new RemoteSembakoRepository(api);
  }

  @After
  public void tearDown() throws Exception {
    server.shutdown();
  }

  @Test
  public void getProductsParsesListAndPath() throws Exception {
    server.enqueue(
        new MockResponse()
            .setResponseCode(200)
            .setBody(
                "[{\"id\":1,\"name\":\"Beras\",\"categoryId\":2,\"supplierId\":3,"
                    + "\"price\":78000,\"stock\":4,\"minStock\":10,\"unit\":\"karung\","
                    + "\"categoryName\":\"Beras & Biji-bijian\","
                    + "\"supplierName\":\"PT Sumber\"}]"));

    List<Product> products = repository.getProducts();

    assertEquals(1, products.size());
    assertEquals("Beras", products.get(0).name);
    assertEquals(1, products.get(0).id);
    assertEquals("/api/products", server.takeRequest().getPath());
  }

  @Test
  public void deleteProductUsesDelete() throws Exception {
    server.enqueue(new MockResponse().setResponseCode(204));

    repository.deleteProduct(7);

    RecordedRequest request = server.takeRequest();
    assertEquals("DELETE", request.getMethod());
    assertEquals("/api/products/7", request.getPath());
  }

  @Test
  public void httpErrorBecomesSembakoException() {
    server.enqueue(new MockResponse().setResponseCode(500));

    SembakoException error = assertThrows(SembakoException.class, repository::getProducts);
    assertTrue(error.getMessage().contains("500"));
  }

  @Test
  public void malformedJsonBecomesSembakoException() {
    server.enqueue(new MockResponse().setResponseCode(200).setBody("not json"));

    assertThrows(SembakoException.class, repository::getProducts);
  }
}
